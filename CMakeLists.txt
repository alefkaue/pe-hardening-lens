cmake_minimum_required(VERSION 3.20)
project(pe_hardening_lens LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(pe_lens src/pe_lens.cpp)
target_include_directories(pe_lens PUBLIC include)
add_executable(pe-hardening-lens src/main.cpp)
target_link_libraries(pe-hardening-lens PRIVATE pe_lens)
add_executable(lens-tests tests/tests.cpp)
target_link_libraries(lens-tests PRIVATE pe_lens)
if(MSVC)
  target_compile_options(pe_lens PRIVATE /W4 /WX)
else()
  target_compile_options(pe_lens PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
option(ENABLE_SANITIZERS "Enable ASan/UBSan" OFF)
if(ENABLE_SANITIZERS AND NOT MSVC)
  foreach(t pe_lens pe-hardening-lens lens-tests)
    target_compile_options(${t} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${t} PRIVATE -fsanitize=address,undefined)
  endforeach()
endif()
enable_testing()
add_test(NAME pe-regressions COMMAND lens-tests)
